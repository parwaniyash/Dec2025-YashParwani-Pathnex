#Ansible Playbook
---
-name:Install and enable Nginx
 hosts:all
 become:yes
 
tasks:
	-name:Install Nginx
	ansible.builtin.apt:
	name:apt
	state:present
	
	-name:Enable Nginx
	ansible.builtin.service:
	name:apt
	state:started
	enabled:yes
