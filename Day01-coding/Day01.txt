#Ansible Playbook
---
- name: Install and enable Nginx
  hosts: all
  become: yes
 
  tasks:
    - name: Install Nginx
      yum:
        name: nginx
        state: present
    - name: Enable Nginx
      service:
        name: nginx
        state: started
        enabled: yes
